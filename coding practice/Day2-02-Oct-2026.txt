Day 02- Service And Tags

#Ansible - Install And Enable Nginx
- name: Install and start Nginx on Pathnex
  hosts: all
  become: yes


tasks:
 - name: Install nginx
   yum:
       name: nginx
       state: present

- name: Enable nginx
  service:
       name: nginx
       state: started
       enables: yes
