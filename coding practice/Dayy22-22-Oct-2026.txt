##Ansible - Deploy Nginx Across Mutiple Kubernetes Clusters

- name: Deploy Nginx across multiple kuberentes clusters
  hosts: localhost
  tasks:
	- name: Configure kubbect1 for cluster 1
	  command: kubect1 --context=us-east config use-context us-east-cluster
	- name: Deploy Nginx on Cluster 1
	  command: kuberct1 apply -f kubernetes/deployment.yaml
	- name: Configure kubect1 for cluster 2
	  command: kubect1 --context=us-west config use-context us-west-cluster
	- name: Deploy Nginx on Cluster 2
	  command: kubectl apply -f kubernetes/deployment.yaml
