Day 07 - Automating Cloud Infrastructure with Terraform &Ansible
##Ansible - Setup Nginx with custom config
- name: Setup Nginx with Custom Config
  hosts: all
  become: yes
  tasks:
	- name: Install nginx
	  yum:
	     name: nginx
	     state: present
	- name: Copy custome nginx.conf
	  copy:
	      src: nginx.conf
	      dest: /etc/nginx/nginx.conf
	- name: Start nginx service
	  service:
	       name: nginx
	       state: started
	       enables: yes

##Terrafrom - Provision EC2 with IAM Role

resource "aws_iam_role" "ec2_role"
{
  name		= "pathnex_ec2_role"
  assume_role_policy 		= jsonencode({	
	Version = "2012-10-17"
	Statement = [
	{
	   Action = "sts:AssumeRole"
	   Effect = "Allow"
	   Principal = {
		Service = "ec2.amazomaws.com"
		}
          },
     ] 
  })
}


resource "aws_instace" "pathnex_ec2"
{
   ami 		= "ami-0abcd1234abc1234"
   instance_type  = "t2.medium"
   iam_instace_profile = aws_iam_role.ec2_role.name
   tags = {
     Name = "Pathnex-EC2"
	}
}
