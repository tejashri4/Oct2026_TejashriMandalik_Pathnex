##Kubernetes - Helm-based Auto-scaling Deployments
apiVersion: apps/v1
kind: Deployment
metadata:
   name: pathnex-web-app
spec:
  replicas: 3
  selector:
     matchLabels:
       app: pathnex-web-app
  template:
      metadata:
       labels: 
          app: pathnex-web-app
   spec:
     containers:
        - name: nginx
          image: nginx
          ports:
		- containerPort: 80

---

apiVerison: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
    name: pathnex-web-app-hpa
spec:
   scaleTargetRef:
      apiVersion: apps/v1
      kind: Deployment
      name: pathnex-web-app
minReplicas: 2
maxReplicas: 10
targetCPUUtilizationPercentage: 80
