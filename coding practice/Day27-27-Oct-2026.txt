#Day 27 - Serverless scaling with lambda and Dyanaomodb

##nsible - deploy serverless Lambada Function

- name:  Deploy Serverless Lambda Function
  hosts: localhost
  tasks:
     - name: Create Lambda fuunction
       command: aws lambda create-funstion --function-name pathnex-lamdba --runtime node.js.x --role arn:aws:iam::12345678901:role/lambda-role --handler index.handler --zip-file fileb://lambda.zip


##Terrafrom - Create DyanmoDB Table for Serverless Application

resource "aws_dynamodb_table" "pathnex_table"
{
 name 		= "pathnex-table"
 hash_key	= "id"
 billing_mode   = "PAY_PER_REQUEST"

   name = "id"
   type = "S"

}
}


##Kubernetes - Deploy Serverless Using Kubeless

apiVersion: kubeless.io/v1beta1
kind: Function
metadata:
   name: pathnex-lambda
   namespace: default
spec: 
   runtime: nodejs14
   handler: index.handler
   function:  |
    module.exports= function(event, context)   {
      context.succesed('Hello World');
};


##Jenkinsfiles - Lambda function deployment pipeline

pipeline{
agent any
stages{
 stage('Build')
{
steps {
echo 'Building Lambda function package'
sh 'zip -r lambda.zip lambda/'
}
}
stage('Deploy to Lambda')
steps{
script{
  sh 'aws lambda update-function-code --function-name pathnex-lambda --zip-file fileb://lambda.zip'
}
}
}
}
}
